-----------------------------------------Kth largest element -----------------------------------------------------------------

#include<iostream>
#include<bits/stdc++.h>
using namespace std;

int getKthLargest(vector<int> &arr, int k){
    priority_queue<int,vector<int>,greater<int>> minHeap;
    for(int i=0; i<arr.size(); i++){
        minHeap.push(arr[i]);
        if(minHeap.size() > k) minHeap.pop();
    }
    return minHeap.top();
}
int main(){
    
    vector<int> arr = {1,3,4,2,5,7,8}; //5
    int k = 3;
    int ans = getKthLargest(arr,k);
    cout << ans << " ";
    return 0;
}
