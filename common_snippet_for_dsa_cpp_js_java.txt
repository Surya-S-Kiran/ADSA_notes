-------------------------------------------C++---------------------------------------------

std::vector<std::pair<int, int>> pairs;
    for (int i = 0; i < num.size(); ++i)
        pairs.push_back({pow[i], num[i]});

std:: sort(pairs.begin(), pairs.end(), [](auto &a, auto &b){
    if(a.first == b.first) return a.second < b.second;  //ascanding order
    return a.first > b.first; //descanding order
});

-------------------------------------------Java---------------------------------------------
List<Pair> list = new ArrayList<>();
    for (int i = 0; i < num.length; i++)
        list.add(new Pair(pow[i], num[i]));

-------------------------------------------JS-----------------------------------------------
let paired = num.map((n, i) => ({ num: n, pow: pow[i] }));

paired.sort((a, b) => {
    if (a.pow === b.pow) return a.num - b.num; // ascending num
    return b.pow - a.pow; // descending pow
});

num = paired.map(p => p.num);
pow = paired.map(p => p.pow);

-------------------------------------------Python---------------------------------------------------------------
paired = list(zip(pow, num))
