int findPivot(int arr[],int n){
    int l = 0, r = n-1;
    while( l < r){
        int m = l + (r-l) / 2;
        if(arr[m] <= arr[r]){
            r = m;
        }else{
            l = m+1;
        }
    }
    return r;
}

 ✅ Iterative Binary Search (Standard)


int binarySearch(vector<int>& arr, int target) {
    int low = 0, high = arr.size() - 1;
    
    while (low <= high) {
        int mid = low + (high - low) / 2;  // avoid overflow
        
        if (arr[mid] == target)
            return mid;
        else if (arr[mid] < target)
            low = mid + 1;
        else
            high = mid - 1;
    }
    return -1; // not found
}

✅ What is Lower Bound and Upper Bound?
Term	Meaning	Example (on sorted arr = [1, 3, 3, 5, 7], target = 3)
Lower Bound	First index ≥ target	1 (arr[1] = 3)
Upper Bound	First index > target	3 (arr[3] = 5)

📌 1. Using STL in C++
C++ STL provides built-in functions:

int upper = upper_bound(arr.begin(), arr.end(), target) - arr.begin();
lower_bound: Returns iterator to first element ≥ target

upper_bound: Returns iterator to first element > target

🧠 Time complexity: O(log n)
